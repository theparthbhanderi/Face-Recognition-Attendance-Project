    document.addEventListener('DOMContentLoaded', function() {
        // Animate user count
        animateValue('totalUsers', 0, 45, 2000);
        animateValue('todayCheckins', 0, 22, 2000);
        
        // Simulate real-time updates
        setInterval(() => {
            const currentUsers = parseInt(document.getElementById('totalUsers').textContent);
            const currentCheckins = parseInt(document.getElementById('todayCheckins').textContent);
            
            // Randomly increment checkins
            if (Math.random() > 0.7) {
                document.getElementById('todayCheckins').textContent = currentCheckins + 1;
            }
        }, 5000);
    });
    
    // Animate counter
    function animateValue(id, start, end, duration) {
        const element = document.getElementById(id);
        const range = end - start;
        const increment = range / (duration / 16);
        let current = start;
        
        const timer = setInterval(() => {
            current += increment;
            if ((increment > 0 && current >= end) || (increment < 0 && current <= end)) {
                current = end;
                clearInterval(timer);
            }
            element.textContent = current;
        }, 16);
    }
</script>

</body>
</html>
