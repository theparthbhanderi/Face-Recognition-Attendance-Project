        // Global Variables
        let video = document.getElementById('video');
        let canvas = document.getElementById('canvas');
        let ctx = canvas.getContext('2d');
        let isRunning = false;
        let startTime = null;
        let currentFaces = [];
        let processedFrames = 0;
        let lastAttendanceMarked = {}; // Track last attendance time per person
        let attendanceCooldown = 300; // 5 minutes cooldown
        let recognitionInterval = null;
        let isProcessingFrame = false;

        // Load initial stats
        async function loadStats() {
            try {
                const response = await fetch('/api/attendance/stats');
                const data = await response.json();
                
                if (data.success) {
                    document.getElementById('totalEntries').textContent = data.data.total_entries;
                    document.getElementById('uniqueUsers').textContent = data.data.unique_users;
                    document.getElementById('knownFaces').textContent = data.data.known_faces_count;
                }
            } catch (error) {
                console.error('Error loading stats:', error);
            }
        }

        // Load daily report
        async function loadDailyReport() {
            try {
                const response = await fetch('/api/attendance/daily_report');
                const data = await response.json();
                
                if (data.success && data.data.length > 0) {
                    const historyList = document.getElementById('historyList');
                    historyList.innerHTML = '';
                    
                    data.data.slice(0, 15).forEach(record => {
                        const time = new Date(record.time).toLocaleTimeString();
                        const item = document.createElement('div');
                        item.className = 'history-item';
                        item.innerHTML = `
                            <strong>${record.name}</strong>
                            <div class="time">${time}</div>
                        `;
                        historyList.appendChild(item);
                    });
                }
            } catch (error) {
                console.error('Error loading daily report:', error);
            }
        }

        // Start Camera - IMPROVED WITH AUTO RECOVERY
        async function startCamera() {
            try {
                const stream = await navigator.mediaDevices.getUserMedia({ 
                    video: { width: 640, height: 480 } 
                });
                
                video.srcObject = stream;
                isRunning = true;
                startTime = Date.now();
                
                document.getElementById('startBtn').disabled = true;
                document.getElementById('stopBtn').disabled = false;
                document.getElementById('status').className = 'status active';
                document.getElementById('status').textContent = '✓ Camera Active - Detecting Faces';
                
                // Update system status indicator
                updateSystemStatus(true);
                
                processVideo();
                updateTimer();
                
                // Start auto-recovery check
                startAutoRecovery();
                
                showNotification('Camera started successfully', 'success');
            } catch (error) {
                showNotification('Error accessing camera: ' + error.message, 'error');
                // Auto-retry after 5 seconds
                setTimeout(() => {
                    if (!isRunning) {
                        showNotification('Retrying camera access...', 'info');
                        startCamera();
                    }
                }, 5000);
            }
        }

        // Stop Camera
        function stopCamera() {
            isRunning = false;
            
            if (video.srcObject) {
                video.srcObject.getTracks().forEach(track => track.stop());
            }
            
            document.getElementById('startBtn').disabled = false;
            document.getElementById('stopBtn').disabled = true;
            document.getElementById('status').className = 'status inactive';
            document.getElementById('status').textContent = 'Not Active';
            
            // Update system status indicator
            updateSystemStatus(false);
            
            stopAutoRecovery();
            showNotification('Camera stopped', 'success');
        }

        // Auto-recovery system
        let recoveryInterval = null;
        function startAutoRecovery() {
            stopAutoRecovery(); // Clear any existing
            recoveryInterval = setInterval(() => {
                if (isRunning && (!video.srcObject || video.readyState === 0)) {
                    console.log('Camera stream lost, attempting recovery...');
                    showNotification('Camera connection lost, recovering...', 'info');
                    stopCamera();
                    setTimeout(() => startCamera(), 1000);
                }
            }, 10000); // Check every 10 seconds
        }

        function stopAutoRecovery() {
            if (recoveryInterval) {
                clearInterval(recoveryInterval);
                recoveryInterval = null;
            }
        }

        // Process video frames - IMPROVED FOR CONTINUOUS OPERATION
        async function processVideo() {
            if (!isRunning) return;

            try {
                // Draw video frame to canvas
                ctx.drawImage(video, 0, 0, canvas.width, canvas.height);

                // Process recognition every frame for better responsiveness
                if (!isProcessingFrame) {
                    isProcessingFrame = true;
                    processRecognitionFrame().finally(() => {
                        isProcessingFrame = false;
                    });
                }

                processedFrames++;
                requestAnimationFrame(processVideo);
            } catch (error) {
                console.error('Video processing error:', error);
                // Continue processing even on errors
                setTimeout(() => requestAnimationFrame(processVideo), 100);
            }
        }

        // Separate recognition processing for better error handling
        async function processRecognitionFrame() {
            try {
                const imageData = canvas.toDataURL('image/jpeg');

                const response = await fetch('/api/recognize_stream', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/x-www-form-urlencoded'
                    },
                    body: 'image_data=' + encodeURIComponent(imageData)
                });

                const result = await response.json();

                if (result.success) {
                    currentFaces = result.results || [];
                    updateDisplayWithFaces();

                    // Auto-mark attendance for recognized faces with cooldown
                    for (const face of currentFaces) {
                        if (face.known && face.user_id) {
                            checkAndMarkAttendance(face.user_id, face.name, face.confidence);
                        }
                    }
                }
            } catch (error) {
                console.error('Recognition processing error:', error);
                // Don't show error notifications to avoid interrupting flow
            }
        }

        // Draw faces on canvas
        function updateDisplayWithFaces() {
            // Clear and redraw video
            ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
            
            // Draw rectangles and labels for detected faces
            currentFaces.forEach(face => {
                const [top, right, bottom, left] = face.location;
                
                // Draw rectangle
                const color = face.known ? '#27ae60' : '#e74c3c';
                ctx.strokeStyle = color;
                ctx.lineWidth = 3;
                ctx.strokeRect(left, top, right - left, bottom - top);
                
                // Draw label background
                ctx.fillStyle = color;
                ctx.fillRect(left, bottom - 30, right - left, 30);
                
                // Draw text
                ctx.fillStyle = 'white';
                ctx.font = 'bold 14px Arial';
                ctx.textAlign = 'center';
                const label = `${face.name} (${face.confidence}%)`;
                ctx.fillText(label, (left + right) / 2, bottom - 10);
            });
            
            // Update recognized list
            updateRecognizedList();
            document.getElementById('faceCountValue').textContent = currentFaces.length;
        }

        // Update recognized persons list - IMPROVED WITH COOLDOWN INFO
        function updateRecognizedList() {
            const recognizedList = document.getElementById('recognizedList');
            const knownFaces = currentFaces.filter(f => f.known);
            
            document.getElementById('recognizedCount').textContent = knownFaces.length;
            
            if (knownFaces.length === 0) {
                recognizedList.innerHTML = '<p style="color: #999; font-size: 12px;">No faces recognized</p>';
                return;
            }
            
            recognizedList.innerHTML = knownFaces.map(face => {
                const now = Date.now();
                const lastMarked = lastAttendanceMarked[face.name];
                const canMark = !lastMarked || (now - lastMarked) >= (attendanceCooldown * 1000);
                
                let statusHtml = '';
                if (!canMark) {
                    const remaining = Math.ceil((attendanceCooldown * 1000 - (now - lastMarked)) / 1000);
                    statusHtml = `<span class="cooldown" style="color: #f39c12; font-size: 11px;">⏱ ${remaining}s</span>`;
                } else {
                    statusHtml = `<span style="color: #27ae60; font-size: 11px;">✓ Ready</span>`;
                }
                
                return `
                    <div class="recognized-person">
                        <span>${face.name}</span>
                        <div style="display: flex; align-items: center; gap: 8px;">
                            <span class="confidence">${face.confidence}%</span>
                            ${statusHtml}
                        </div>
                    </div>
                `;
            }).join('');
        }

        // Check and mark attendance with cooldown
        async function checkAndMarkAttendance(userId, name, confidence) {
            const now = Date.now();
            const lastMarked = lastAttendanceMarked[name];

            if (lastMarked && (now - lastMarked) < (attendanceCooldown * 1000)) {
                // Still in cooldown period - show remaining time subtly
                const remaining = Math.ceil((attendanceCooldown * 1000 - (now - lastMarked)) / 1000);
                updateCooldownDisplay(name, remaining);
                return;
            }

            // Mark attendance
            try {
                const response = await fetch('/api/mark_attendance_auto', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify({ user_id: userId, name: name })
                });

                const result = await response.json();

                if (result.success) {
                    lastAttendanceMarked[name] = now;
                    showSuccessNotification(`✓ ${name} (${confidence}%)`, false); // Less intrusive
                    loadStats();
                    loadDailyReport();
                    clearCooldownDisplay(name);
                } else {
                    console.log(`Attendance: ${result.message}`);
                }
            } catch (error) {
                console.error('Error marking attendance:', error);
            }
        }

        // Update cooldown display
        function updateCooldownDisplay(name, remainingSeconds) {
            const recognizedList = document.getElementById('recognizedList');
            const personElements = recognizedList.querySelectorAll('.recognized-person');

            personElements.forEach(element => {
                if (element.textContent.includes(name)) {
                    let cooldownSpan = element.querySelector('.cooldown');
                    if (!cooldownSpan) {
                        cooldownSpan = document.createElement('span');
                        cooldownSpan.className = 'cooldown';
                        cooldownSpan.style.cssText = 'color: #f39c12; font-size: 11px; margin-left: 10px;';
                        element.appendChild(cooldownSpan);
                    }
                    cooldownSpan.textContent = `⏱ ${remainingSeconds}s`;
                }
            });
        }

        // Clear cooldown display
        function clearCooldownDisplay(name) {
            const recognizedList = document.getElementById('recognizedList');
            const personElements = recognizedList.querySelectorAll('.recognized-person');

            personElements.forEach(element => {
                if (element.textContent.includes(name)) {
                    const cooldownSpan = element.querySelector('.cooldown');
                    if (cooldownSpan) {
                        cooldownSpan.remove();
                    }
                }
            });
        }

        // Update system status indicator
        function updateSystemStatus(active) {
            const indicator = document.getElementById('systemStatus');
            if (active) {
                indicator.style.background = '#27ae60';
                indicator.style.display = 'inline-block';
            } else {
                indicator.style.background = '#e74c3c';
                indicator.style.display = 'inline-block';
            }
        }

        // Enhanced error recovery
        window.addEventListener('error', function(e) {
            console.error('Global error:', e.error);
            // Don't interrupt the flow, just log
        });

        // Handle visibility change to maintain operation
        document.addEventListener('visibilitychange', function() {
            if (document.hidden) {
                console.log('Page hidden, maintaining operation...');
            } else {
                console.log('Page visible, checking camera status...');
                if (isRunning && (!video.srcObject || video.readyState === 0)) {
                    console.log('Camera lost during page hide, restarting...');
                    startCamera();
                }
            }
        });

        // Update timer
        function updateTimer() {
            if (!isRunning) return;
            
            const elapsed = Math.floor((Date.now() - startTime) / 1000);
            const minutes = Math.floor(elapsed / 60);
            const seconds = elapsed % 60;
            
            document.getElementById('timer').textContent = 
                `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
            
            setTimeout(updateTimer, 1000);
        }

        // Show notification - IMPROVED FOR LESS INTRUSION
        function showNotification(message, type, intrusive = true) {
            if (!intrusive && type === 'success') {
                // For success messages, show a subtle indicator instead of popup
                showSuccessNotification(message, false);
                return;
            }

            const notification = document.createElement('div');
            notification.className = `notification ${type}`;
            notification.textContent = message;

            document.body.appendChild(notification);

            setTimeout(() => {
                notification.style.opacity = '0';
                setTimeout(() => notification.remove(), 300);
            }, 3000);
        }

        // Subtle success notification
        function showSuccessNotification(message, showPopup = true) {
            if (showPopup) {
                showNotification(message, 'success', true);
                return;
            }

            // Update status bar with success message
            const status = document.getElementById('status');
            const originalText = status.textContent;
            status.textContent = message;
            status.style.color = '#27ae60';
            status.style.fontWeight = 'bold';

            setTimeout(() => {
                status.textContent = originalText;
                status.style.color = '';
                status.style.fontWeight = '';
            }, 2000);
        }

        // Periodically refresh stats and history
        setInterval(loadStats, 10000); // Every 10 seconds
        setInterval(loadDailyReport, 5000); // Every 5 seconds

        // Load initial data
        loadStats();
        loadDailyReport();

        // Cleanup on page unload
        window.addEventListener('beforeunload', () => {
            stopCamera();
        });
